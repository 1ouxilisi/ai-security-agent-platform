💻 Python SDK 示例

from aihacking import AIHackingClient

# 初始化客户端
client = AIHackingClient(api_key="sk-your-api-key")

# 创建扫描任务
scan = client.scans.create(
    target="http://example.com",
    scan_type="full",
    severity="high"
)
print(f"扫描ID: {scan.id}")

# 等待扫描完成
result = client.scans.wait(scan.id, timeout=300)
print(f"发现漏洞: {len(result.vulnerabilities)}")

# 生成报告
report = client.reports.create(
    scan_id=scan.id,
    template="standard",
    format="markdown"
)
print(f"报告: {report.download_url}")

# 查询CVE
cve = client.knowledge.get_cve("CVE-2021-44228")
print(f"漏洞: {cve.name}, 评分: {cve.cvss}")
