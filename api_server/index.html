⚡ 常用API速查（直接复制用）
# 健康检查
curl http://127.0.0.1:8000/health
# 搜索漏洞
curl "http://127.0.0.1:8000/api/v1/vuln/search?keyword=Log4j&severity=critical"
# 同步NVD（后台）
curl -X POST http://127.0.0.1:8000/api/v1/vuln/sync/nvd -H "Content-Type: application/json" -d '{"days":30}'
# 调用工具（端口扫描）
curl -X POST http://127.0.0.1:8000/api/v1/tools/call -H "Content-Type: application/json" -d '{"tool_name":"port_scan","parameters":{"target":"127.0.0.1"}}'
# 创建扫描任务
curl -X POST http://127.0.0.1:8000/api/v1/tasks -H "Content-Type: application/json" -d '{"target":"127.0.0.1","task_type":"full_scan"}'
# AI智能体执行
curl -X POST http://127.0.0.1:8000/api/v1/agents/react -H "Content-Type: application/json" -d '{"task":"扫描127.0.0.1的开放端口"}'
# 超级智能体（全自动渗透）
curl -X POST http://127.0.0.1:8000/api/v1/super-agent/execute -H "Content-Type: application/json" -d '{"target":"http://靶场地址","goal":"全面渗透测试并生成报告"}'
# 生成报告
curl -X POST http://127.0.0.1:8000/api/v1/reports/generate -H "Content-Type: application/json" -d '{"task_id":"任务ID","format":"markdown"}'
